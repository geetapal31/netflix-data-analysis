def catigorize_col(df, col, lebels):
    desc = df[col].describe()
    edges = [desc['min'], desc['25%'], desc['50%'], desc['75%'], desc['max']]
    df[col] = pd.cut(df[col], bins=edges, labels=labels, include_lowest=True)
    return df
